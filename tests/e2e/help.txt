aub: one ledger for LLM consumption

usage: aub [-v...] <command> [--format text|json] [options]
shared flags: -v, --format text|json, --explain[=summary|full], --account NAME, --model MODEL, --no-color

commands:
  status   render the last known meter reading per configured account
  spend    canonical token usage grouped by day, session, project or repository
  config   print every resolved configuration key with the source that won
  export   emit versioned JSONL usage rows keyed by session-id or run-id for external joins
  rate-card import, show and history the immutable dated vendor rate cards
  backup   create or verify a consistent archive of durable state
  ingest   land the configured transcripts' normalized usage into the ledger, advancing the ingestion generation
  rebuild  destroy and recreate one rebuildable materialization group, never touching irreplaceable evidence
  doctor   health, drift and integrity diagnostics
  coverage did the sampler attempt what the policy owed, and did those attempts observe?
  import   import explicitly selected legacy evidence with durable provenance

status:
  answers: how much quota does each configured account have left?
  refuses: --no-color (status prints no color)
  format: text | json

spend:
  answers: how many canonical tokens were used, grouped by the requested dimensions?
  refuses: --account (spend has no account dimension until account attribution lands); --model (spend has no model dimension until a model selector is needed); --no-color (spend prints no color)
  format: text | json
  options: --today (default) | --since YYYY-MM-DD | --days N | --group-by day|session|project|repository (repeatable) | --refresh auto|never|force

config:
  answers: which configuration key resolved from where?
  refuses: --format (config prints provenance, not a report); --explain (config derives no quantity); --account (config prints every account at once); --model (config prints every key at once); --no-color (config prints no color)
  format: text only: config prints provenance, not a report
  options: --set key=value (repeatable), --config-file PATH

export:
  answers: which usage did each session or run consume, as a versioned JSONL ledger for an external join?
  refuses: --format (export is always versioned JSONL); --explain (export derives no quantity); --account (export reads every stored session, not one account); --model (export keys on session or run, not on a model); --no-color (export prints no color)
  format: text only: export is always versioned JSONL
  options: --key session-id|run-id (required), --include-logical-ids

rate-card:
  answers: what do the immutable dated vendor rate cards contain?
  refuses: --format (rate-card prints a price book, not a report); --explain (rate-card derives no quantity); --account (the rate book is reference data, not per-account state); --model (the rate book is reference data, not per-model state); --no-color (rate-card prints plain rows)
  format: text only: rate-card prints a price book, not a report

backup:
  answers: is there a consistent, verified archive of the durable state, and does it restore?
  refuses: --format (backup prints one operational result); --explain (backup derives no quantity); --account (backup covers the whole ledger); --model (backup covers the whole ledger); --no-color (backup prints no color)
  format: text only: backup prints one operational result
  options: DESTINATION | verify DESTINATION

ingest:
  answers: have the transcript-derived tables been refreshed from the transcripts on disk, under one generation?
  refuses: --format (ingest prints one operational result); --explain (ingest derives no quantity); --account (ingest reads the configured transcript sources); --model (ingest reads transcript sources, not models); --no-color (ingest prints no color)
  format: text only: ingest prints one operational result
  options: transcripts [--source NAME] [--changed-only]

rebuild:
  answers: can the transcript-derived materializations be rebuilt from scratch while every irreplaceable record is left untouched?
  refuses: --format (rebuild prints one operational result); --explain (rebuild derives no quantity); --account (rebuild sweeps whole materialization tables, not one account); --model (rebuild sweeps whole materialization tables, not models); --no-color (rebuild prints no color)
  format: text only: rebuild prints one operational result
  options: transcripts | attribution

doctor:
  answers: is the recorded evidence healthy, and does the transcript corpus still match its parsers?
  refuses: --explain (doctor derives no quantity); --account (doctor is a system-wide diagnostic); --model (doctor is a system-wide diagnostic); --no-color (doctor prints plain text or json)
  format: text | json
  options: --transcript-format-drift

coverage:
  answers: did the sampler attempt what the policy owed, and did those attempts observe?
  refuses: --model (coverage reports every window of the account, not one model); --no-color (coverage prints plain text or json)
  format: text | json
  options: --since DURATION (default 24h), --severe; --account is shared

import:
  answers: which legacy evidence is safe to import into the ledger?
  refuses: --format (import prints one operational result); --explain (import derives no report); --account (import carries account identity in its source); --model (import carries no model selector); --no-color (import prints no color)
  format: text only: import prints one operational result
  options: legacy-meter --source PATH --backup VERIFIED_ARCHIVE

aub            prints the version; aub --help prints this
